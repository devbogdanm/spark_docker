------compare csv vs json vs parquet

data = [(i, f"name_{i}", i*100, "2026-11-01") for i in range(1_000_000)]
columns = ["id","name","salary","date"]
df = spark.createDataFrame(data,columns)

df.write.mode("overwrite").option("header","true").csv("/opt/spark-data/employees_csv")
df.write.mode("overwrite").json("/opt/spark-data/employees_json")
df.write.mode("overwrite").parquet("/opt/spark-data/employees_parquet")

--csv.(path) insted of format(csv) and load(path)

 parquet_df.select("name").count()

from pyspark.sql.types import(StructType, StructField, IntegerType, StringType, DoubleType)
schema = StructType([StructField("id", IntegerType(), True), StructField("name", StringType(), True), StructField("points", DoubleType(), True)])
df_explicit = spark.read.option("header", "true").schema(schema).csv("/opt/spark-data/schema_test")
df_explicit.printSchema()
df_explicit.show()
